cmake_minimum_required(VERSION 3.16)
project(ProvinceRoadNetwork LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(routing src/Map.cpp src/GPS.cpp)
target_include_directories(routing PUBLIC include)
if(MSVC)
  target_compile_options(routing PRIVATE /W4 /utf-8)
else()
  target_compile_options(routing PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(app src/main.cpp src/app.cpp)
target_link_libraries(app PRIVATE routing)
enable_testing()
add_executable(route-tests tests/route-tests.cpp)
target_link_libraries(route-tests PRIVATE routing)
add_test(NAME routing COMMAND route-tests)
